// Android · Kotlin · Coroutines + Flow
class FeedViewModel(private val repo: FeedRepository) : ViewModel() {

    private val _state = MutableStateFlow(FeedState.Loading)
    val state: StateFlow<FeedState> = _state.asStateFlow()

    fun refresh() = viewModelScope.launch {
        _state.value = runCatching { repo.feed() }
            .fold(::FeedState.Ready, ::FeedState.Error)
    }
}
 
// Backend · Java 17 · Spring Boot 3
@RestController
@RequestMapping("/api/v1/feed")
class FeedController(private final FeedService service) {

    @GetMapping
    @PreAuthorize("hasRole('USER')")
    public Page<FeedItem> feed(Pageable page) {
        return service.cachedFeed(page); // Redis-backed
    }
}
